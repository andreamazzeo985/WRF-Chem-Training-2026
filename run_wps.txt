# ---------------------------------------
# 1. Load WPS environment
# ---------------------------------------

# Use the independent course installation
source /home/hpc/46/mazzeo/storage/pgm/Models/WRF-Chem_Course_Student/set_env.sh

# ---------------------------------------
# 2. Define directories
# ---------------------------------------

export WPS_DIR="/home/hpc/46/mazzeo/storage/pgm/Models/WRF-Chem_Course_Student/WPS-4.5"

# Directory containing namelist.wps,
# GRIBFILE links and Vtable
export RUN_DIR="/home/hpc/46/mazzeo/scratch/wrf-chem/simuldir/WPS/"
export NAMELIST_DIR="/home/hpc/46/mazzeo/scratch/wrf-chem/simuldir/WPS/"

cd "$RUN_DIR"

# ---------------------------------------
# 3. Check executables and input files
# ---------------------------------------

for exe in geogrid.exe ungrib.exe metgrid.exe; do
    if [ ! -x "$WPS_DIR/$exe" ]; then
        echo "ERROR: Missing executable: $WPS_DIR/$exe"
        exit 1
    fi
done

echo "Preparing WPS simulation directory..."

# Copy WPS configuration
if [ ! -f namelist.wps ]; then
    cp "$WPS_DIR/namelist.wps" .
fi

# Link the WPS executables
for exe in geogrid.exe ungrib.exe metgrid.exe; do
    if [ ! -x "$WPS_DIR/$exe" ]; then
        echo "ERROR: Missing WPS executable: $WPS_DIR/$exe"
        exit 1
    fi
    ln -sfn "$WPS_DIR/$exe" "$RUN_DIR/$exe"
done

ln -sfn "$WPS_DIR/geogrid" geogrid
ln -sfn "$WPS_DIR/ungrib" ungrib
ln -sfn "$WPS_DIR/metgrid" metgrid

echo "WPS directory prepared."

echo "========================================"
echo "WPS COURSE PREPROCESSING"
echo "========================================"
echo "Job ID: ${SLURM_JOB_ID}"
echo "Node: $(hostname)"
echo "MPI tasks: ${SLURM_NTASKS}"
echo "WPS directory: $WPS_DIR"
echo "Run directory: $RUN_DIR"
echo "Start: $(date)"
echo "========================================"

# ---------------------------------------
# 4. GEOGRID
# ---------------------------------------

echo "Starting GEOGRID: $(date)"
